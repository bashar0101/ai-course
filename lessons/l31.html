  • واحد: نموذج مثالي، بلا أخطاء.

setup.pyبايثون
import numpy as np
from sklearn.datasets import load_diabetes
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.dummy import DummyRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

X, y = load_diabetes(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.25, random_state=0)
print("target range:", y.min(), "to", y.max(), " mean:", round(y.mean(), 1))

def report(name, model):
    p = model.fit(X_tr, y_tr).predict(X_te)
    mae = mean_absolute_error(y_te, p)
    rmse = np.sqrt(mean_squared_error(y_te, p))
    print(f"{name:14} MAE = {mae:6.1f}   RMSE = {rmse:6.1f}   R2 = {r2_score(y_te, p):6.3f}")
    return p

report("mean baseline", DummyRegressor(strategy="mean"))
p_lin = report("linear", LinearRegression())
report("forest", RandomForestRegressor(n_estimators=300, random_state=0))

سؤال مقابلة

سؤال قد تسمعه في المقابلة:

What is the difference between MAE and RMSE, and what does R² tell you?

جواب قصير وجيد:

MAE is the average absolute error; it is easy to explain and robust to outliers. RMSE squares errors before averaging, so large errors count much more; it is always at least as large as MAE, and a big gap means a few large errors. R² compares the model's squared error with always predicting the mean: 1 is perfect, 0 is no better than the mean, negative is worse.

الخلاصة
  • متوسط الخطأ المطلق: سهل الشرح، وهادئ مع القيم الشاذة.
  • متوسط المربعات وجذره: يعاقبان الأخطاء الكبيرة بشدة. الجذر بوحدة الجواب.
  • معامل التحديد: كم أفضل من توقع المتوسط دائماً. صفر يعني لا فائدة، وسالب يعني أسوأ.
  • الفرق الكبير بين الجذر ومتوسط الخطأ المطلق علامة أخطاء كبيرة قليلة.
  • قارن دائماً بنموذج المتوسط، وعلى بيانات الاختبار.
