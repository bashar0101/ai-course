مشكلتان متعاكستان

قلة التعلّم: النموذج بسيط جداً على المشكلة. مثل خط مستقيم لبيانات على شكل قوس.

يخطئ كثيراً على بيانات التدريب نفسها. لم يتعلّم حتى ما رآه.

الإفراط في التعلّم: النموذج معقد جداً على كمية البيانات. يحفظ كل نقطة، ومعها الضجيج.

ممتاز على التدريب، وسيئ على البيانات الجديدة. مثل الطالب الذي حفظ الأجوبة.

complexity.pyبايثون
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import PolynomialFeatures, StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

rng = np.random.default_rng(0)
x = rng.uniform(-1, 1, 60)
y = np.sin(2.6 * x) * 1.2 + 0.4 * x + rng.normal(0, 0.28, x.size)
X = x.reshape(-1, 1)
X_tr, X_val, y_tr, y_val = train_test_split(X, y, test_size=0.4, random_state=0)

degrees = range(0, 16)
tr_err, val_err = [], []
for d in degrees:
    m = make_pipeline(PolynomialFeatures(d), StandardScaler(), LinearRegression()).fit(X_tr, y_tr)
    tr_err.append(mean_squared_error(y_tr, m.predict(X_tr)))
    val_err.append(mean_squared_error(y_val, m.predict(X_val)))

plt.figure(figsize=(6, 3.4))
plt.plot(degrees, tr_err, "o-", lw=2, label="train")
plt.plot(degrees, val_err, "o-", lw=2, label="validation")
plt.yscale("log")
plt.xlabel("polynomial degree (model complexity)"); plt.ylabel("MSE (log scale)")
plt.legend(); plt.grid(True)
plt.show()

best = int(np.argmin(val_err))
print("best degree by validation:", best, " train MSE:", round(tr_err[best], 3), " val MSE:", round(val_err[best], 3))

from sklearn.model_selection import learning_curve
from sklearn.datasets import load_digits
from sklearn.tree import DecisionTreeClassifier
from sklearn.linear_model import LogisticRegression

Xd, yd = load_digits(return_X_y=True)
fig, axes = plt.subplots(1, 2, figsize=(10, 3.4), sharey=True)
for ax, (name, model) in zip(axes, [("deep tree", DecisionTreeClassifier(random_state=0)),
                                     ("logistic regression", make_pipeline(StandardScaler(), LogisticRegression(max_iter=2000)))]):
    sizes, tr, va = learning_curve(model, Xd, yd, cv=5, train_sizes=np.linspace(0.1, 1.0, 6))
    ax.plot(sizes, tr.mean(axis=1), "o-", lw=2, label="train accuracy")
    ax.plot(sizes, va.mean(axis=1), "o-", lw=2, label="validation accuracy")
    ax.set_title(name); ax.set_xlabel("training examples"); ax.grid(True)
axes[0].set_ylabel("accuracy"); axes[0].legend()
plt.tight_layout()
plt.show()

كيف تقرأ منحنى التعلّم:

  • فجوة كبيرة، والتحقق ما زال يصعد: إفراط. بيانات أكثر ستساعد.
  • الخطان التقيا ويستقران على أداء سيئ: قلة تعلّم. بيانات أكثر لن تساعد. تحتاج نموذجاً أقوى أو ميزات أفضل.

سؤال مقابلة

سؤال قد تسمعه في المقابلة:

Explain the bias-variance tradeoff.

جواب قصير وجيد:

Bias is the error from wrong or too simple assumptions: a simple model misses real patterns and underfits. Variance is the error from being too sensitive to the specific training sample: a complex model fits the noise and overfits. Making a model more complex usually lowers bias and raises variance. We choose the complexity that gives the lowest error on validation data.

الخلاصة
  • قلة التعلّم: خطأ كبير في التدريب والتحقق. الإفراط: تدريب ممتاز، وتحقق سيئ.
  • الانحياز من البساطة الزائدة. والتباين من الحساسية الزائدة لعينة التدريب.
  • خطأ التدريب ينزل دائماً مع التعقيد. خطأ التحقق ينزل ثم يصعد. نختار القاع.
  • منحنى التعلّم يقول هل البيانات الإضافية ستساعد.
  • علاج الإفراط: بيانات أكثر، وتبسيط، وتنظيم. وعلاج قلة التعلّم: نموذج أقوى وميزات أفضل.

from sklearn.datasets import load_breast_cancer
from sklearn.tree import DecisionTreeClassifier

Xb, yb = load_breast_cancer(return_X_y=True)
X_tmp, X_te, y_tmp, y_te = train_test_split(Xb, yb, test_size=0.2, random_state=1, stratify=yb)
X_tr, X_va, y_tr, y_va = train_test_split(X_tmp, y_tmp, test_size=0.25, random_state=1, stratify=y_tmp)

depths = range(1, 13)
tr_acc, va_acc = [], []
for d in depths:
    t = DecisionTreeClassifier(max_depth=d, random_state=0).fit(X_tr, y_tr)
    tr_acc.append(t.score(X_tr, y_tr))
    va_acc.append(t.score(X_va, y_va))

plt.figure(figsize=(6, 3.2))
plt.plot(depths, tr_acc, "o-", label="train"); plt.plot(depths, va_acc, "o-", label="validation")
plt.xlabel("max_depth"); plt.legend(); plt.grid(True)
plt.show()

best = depths[int(np.argmax(va_acc))]
final = DecisionTreeClassifier(max_depth=best, random_state=0).fit(X_tr, y_tr)
print("best depth:", best, " validation:", round(max(va_acc), 3), " test:", round(final.score(X_te, y_te), 3))
